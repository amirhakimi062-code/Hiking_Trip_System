CREATE TABLE CUSTOMER (
    customer_id VARCHAR2(10) PRIMARY KEY,
    customer_name VARCHAR2(100) NOT NULL,
    customer_phone VARCHAR2(20) NOT NULL,
    customer_email VARCHAR2(100) NOT NULL UNIQUE,
    customer_pass VARCHAR2(50) NOT NULL
);

CREATE TABLE STAFF (
    staff_id VARCHAR2(10) PRIMARY KEY,
    staff_name VARCHAR2(100) NOT NULL,
    staff_phone VARCHAR2(20) NOT NULL,
    staff_email VARCHAR2(100) NOT NULL UNIQUE,
    staff_pass VARCHAR2(50) NOT NULL
);

CREATE TABLE LOCATION_HIKING (
    location_id VARCHAR2(10) PRIMARY KEY,
    location_name VARCHAR2(100) NOT NULL,
    price_per_pax NUMBER(10, 2) NOT NULL
);

CREATE TABLE GUIDE (
    guide_id VARCHAR2(10) PRIMARY KEY,
    guide_name VARCHAR2(100) NOT NULL,
    guide_phone VARCHAR2(20) NOT NULL,
    guide_email VARCHAR2(100) NOT NULL UNIQUE,
    guide_pass VARCHAR2(50) NOT NULL
);

CREATE TABLE PAYMENT (
    payment_id VARCHAR2(10) PRIMARY KEY,
    payment_method VARCHAR2(50) NOT NULL CHECK (payment_method IN ('QR', 'TnG', 'ONLINE TRANSFER')),
    total_amount NUMBER(10, 2) NOT NULL,
    payment_date DATE NOT NULL,
    guide_fee NUMBER(10, 2) DEFAULT 50.00 NOT NULL
);

CREATE TABLE BOOKING (
    booking_id VARCHAR2(10) PRIMARY KEY,
    customer_id VARCHAR2(10) NOT NULL,
    staff_id VARCHAR2(10) NOT NULL,
    booking_date DATE NOT NULL,
    CONSTRAINT fk_booking_customer FOREIGN KEY (customer_id) REFERENCES CUSTOMER(customer_id),
    CONSTRAINT fk_booking_staff FOREIGN KEY (staff_id) REFERENCES STAFF(staff_id)
);

CREATE TABLE BOOKING_DETAIL (
    booking_id VARCHAR2(10) NOT NULL,
    location_id VARCHAR2(10) NOT NULL,
    payment_id VARCHAR2(10) NOT NULL,
    guide_id VARCHAR2(10) NOT NULL,
    pax NUMBER NOT NULL,
    subtotal NUMBER(10, 2) NOT NULL,
    duration_day NUMBER NOT NULL CHECK (duration_day BETWEEN 1 AND 7),
    permit_fee NUMBER(10, 2) DEFAULT 50.00 NOT NULL,
    equipment_fee NUMBER(10, 2),          
    emergency_contact VARCHAR2(50),       
    PRIMARY KEY (booking_id, location_id),
    CONSTRAINT fk_detail_booking FOREIGN KEY (booking_id) REFERENCES BOOKING(booking_id),
    CONSTRAINT fk_detail_location FOREIGN KEY (location_id) REFERENCES LOCATION_HIKING(location_id),
    CONSTRAINT fk_detail_payment FOREIGN KEY (payment_id) REFERENCES PAYMENT(payment_id),
    CONSTRAINT fk_detail_guide FOREIGN KEY (guide_id) REFERENCES GUIDE(guide_id)
);


-- ==========================================
-- 2. SEQUENCE & AUTOMATIC TRIGGERS
-- ==========================================

CREATE SEQUENCE seq_customer_id START WITH 1 INCREMENT BY 1;
/
CREATE OR REPLACE TRIGGER trg_customer_id
BEFORE INSERT ON CUSTOMER FOR EACH ROW
BEGIN
    IF :NEW.customer_id IS NULL THEN
        :NEW.customer_id := 'C' || LPAD(seq_customer_id.NEXTVAL, 3, '0');
    END IF;
END;
/

CREATE SEQUENCE seq_staff_id START WITH 1 INCREMENT BY 1;
/
CREATE OR REPLACE TRIGGER trg_staff_id
BEFORE INSERT ON STAFF FOR EACH ROW
BEGIN
    IF :NEW.staff_id IS NULL THEN
        :NEW.staff_id := 'S' || LPAD(seq_staff_id.NEXTVAL, 3, '0');
    END IF;
END;
/

CREATE SEQUENCE seq_guide_id START WITH 1 INCREMENT BY 1;
/
CREATE OR REPLACE TRIGGER trg_guide_id
BEFORE INSERT ON GUIDE FOR EACH ROW
BEGIN
    IF :NEW.guide_id IS NULL THEN
        :NEW.guide_id := 'G' || LPAD(seq_guide_id.NEXTVAL, 3, '0');
    END IF;
END;
/

CREATE SEQUENCE seq_location_id START WITH 1 INCREMENT BY 1;
/
CREATE OR REPLACE TRIGGER trg_location_id
BEFORE INSERT ON LOCATION_HIKING FOR EACH ROW
BEGIN
    IF :NEW.location_id IS NULL THEN
        :NEW.location_id := 'L' || LPAD(seq_location_id.NEXTVAL, 3, '0');
    END IF;
END;
/

CREATE SEQUENCE seq_payment_id START WITH 1 INCREMENT BY 1;
/
CREATE OR REPLACE TRIGGER trg_payment_id
BEFORE INSERT ON PAYMENT FOR EACH ROW
BEGIN
    IF :NEW.payment_id IS NULL THEN
        :NEW.payment_id := 'P' || LPAD(seq_payment_id.NEXTVAL, 3, '0');
    END IF;
END;
/

CREATE SEQUENCE seq_booking_id START WITH 1 INCREMENT BY 1;
/
CREATE OR REPLACE TRIGGER trg_booking_id
BEFORE INSERT ON BOOKING FOR EACH ROW
BEGIN
    IF :NEW.booking_id IS NULL THEN
        :NEW.booking_id := 'B' || LPAD(seq_booking_id.NEXTVAL, 3, '0');
    END IF;
END;
/

CREATE OR REPLACE TRIGGER trg_booking_detail_fee
BEFORE INSERT ON BOOKING_DETAIL FOR EACH ROW
BEGIN
    IF :NEW.equipment_fee IS NULL THEN
        :NEW.equipment_fee := :NEW.pax * 40;
    END IF;
END;
/
